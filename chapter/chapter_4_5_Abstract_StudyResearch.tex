	\renewcommand{\abstractname}{Reminder}
	\begin{abstract}
			\begin{description}
			\item[Study] The process of learning or understanding a subject, typically through reading, lectures, and consuming existing information. It is foundational and inward-focused on learning what is already known.
			\item[Research]  The active, systematic search for new knowledge to answer specific questions or solve problems. It goes beyond existing knowledge to create, discover, or innovate. Research often uses studies (e.g., case studies, literature studies) as part of its methodology.
			\item[Science] A structured, empirical, and, most importantly, reproducible system for acquiring knowledge about the universe through observation and experimentation. While research can be in any field (e.g., historical, market research), science is specific to building knowledge that is generally objective and testable. 
		\end{description}
	\end{abstract}
}
